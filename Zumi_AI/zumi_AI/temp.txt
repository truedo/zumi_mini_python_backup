    def __sketchdetect(self):
        while self.__sketchDetectFlag:
            if self.__raw_img is None:
                time.sleep(0.1)
                # print('no input frame yet')
                continue
            try:
                self.__sketchRecognizedList, self.__sketchDetectedList = self.__sketchR(self.__raw_img)
                self.__sketchDataDict.clear()
                for i in range(0, len(self.__sketchDetectedList)):
                    self.__sketchDataDict[self.__sketchRecognizedList[i]] = SketchData( self.__sketchRecognizedList[i], self.__sketchDetectedList[i])

                if len(self.__sketchRecognizedList) == 0:
                    time.sleep(0.0)
                    continue
            except Exception as e:
                # print("Sketch detector error : " , e)
                continue

            time.sleep(0.01)

    def SketchCapture(self, name:str, captureCount:int=5, path:str=pkg_resources.resource_filename(__package__,"res/sketch/")):
        if bool(name) == False:
            print("Name parameter is Empty.")
            return

        if os.path.isdir(path) is False:
            os.mkdir(path)

        if self.__sketchDetectFlag is False:
            print("Sketchdetector did not run")
            return

        cnt = 0
        while cnt < captureCount:
            if len(self.__sketchRecognizedList) == 0:
                print("Doesn't have a any sketch in Frame")
                time.sleep(0.0)
                continue

            result = self.__sketchR.SaveSketch(self.__raw_img,name)
            if result == 0:
                cnt += 1
                time.sleep(0.1)
        print( name, " is saved")

    def TrainSketchData(self, sketchPath:str = pkg_resources.resource_filename(__package__,"res/sketch/")):
        if os.path.isdir(sketchPath) is False:
            print(sketchPath +" is not directory.")
            return

        orbDescriptors = []
        nameIndexList = []
        nameIntList = []

        sketchD = SketchRecognizer()
        filenames = os.listdir(sketchPath)
        for filename in filenames:
            name = os.path.basename(filename)
            image = cv2.imread(sketchPath+filename, cv2.IMREAD_GRAYSCALE)
            image = cv2.resize(image, (150,150))
            _, des = sketchD.orbDetector.detectAndCompute(image, None)
            name = name.split('_')[0]

            if not(name in nameIndexList):
                nameIndexList.append(name)
            nameIntList.append(nameIndexList.index(name))
            orbDescriptors.append(des)

        self.__sketchR.TrainModel(nameIndexList, nameIntList, orbDescriptors)

    def DeleteSketchData(self, name:str, sketchPath:str=pkg_resources.resource_filename(__package__,"res/sketch/")):

        if os.path.isdir(sketchPath) is False:
            print(sketchPath +" is not directory.")
            return

        self.__sketchR.RemoveSketch(name, sketchPath)

        print(name + ' is deleted')

        # ROI 면적 필터링 (픽셀 단위, 영상 해상도에 따라 조절)
        MIN_ROI_AREA = 100 # 최소 표지판 면적 (너무 작은 노이즈 제거)
        MAX_ROI_AREA = 10000 # 최대 표지판 면적 (너무 큰 배경 객체 제거)

        # 모폴로지 연산 커널
        MORPH_KERNEL = np.ones((3, 3), np.uint8) # 커널 크기 조정 가능

        # Hough Circle 파라미터 (이전 코드와 동일, 필요시 튜닝)
        HOUGH_DP = 1.2
        HOUGH_MIN_DIST = 50

        HOUGH_PARAM2 = 30
        HOUGH_MIN_RADIUS = 15

        while self.__signDetectFlag:

                # mask_blue = cv2.inRange(hsv, lower_blue, upper_blue)

                # # 두 색상 마스크 합치기
                # combined_mask = cv2.bitwise_or(mask_red, mask_blue)

                # # combined_mask = cv2.morphologyEx(combined_mask, cv2.MORPH_CLOSE, kernel) # 닫기 (작은 구멍 메우기)

                # # 윤곽선 찾기
                # contours, _ = cv2.findContours(combined_mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

                #         # 경계 상자 얻기

                #         # 종횡비 필터링 (표지판은 대략 정사각형 또는 원형에 가까움)

                #             self.__signDetectedRegions.append((x, y, w, h))

                hsv = cv2.cvtColor(self.__raw_img, cv2.COLOR_BGR2HSV)

                # --- 1. 색상 기반 마스크 생성 ---
                # 빨간색 마스크
                mask_red1 = cv2.inRange(hsv, lower_red1, upper_red1)
                mask_red2 = cv2.inRange(hsv, lower_red2, upper_red2)
                mask_red = cv2.bitwise_or(mask_red1, mask_red2)

                # 파란색 마스크
                mask_blue = cv2.inRange(hsv, lower_blue, upper_blue)

                # 노란색 마스크 (추가)
                mask_yellow = cv2.inRange(hsv, lower_yellow, upper_yellow)

                # 모든 색상 마스크 합치기
                combined_mask = cv2.bitwise_or(mask_red, mask_blue)
                combined_mask = cv2.bitwise_or(combined_mask, mask_yellow)

                # --- 2. 모폴로지 연산 (노이즈 제거 및 윤곽선 연결) ---
                # 열기: 작은 노이즈 제거
                combined_mask = cv2.morphologyEx(combined_mask, cv2.MORPH_OPEN, MORPH_KERNEL, iterations=1)
                # 닫기: 작은 구멍 메우기, 끊어진 윤곽선 연결
                combined_mask = cv2.morphologyEx(combined_mask, cv2.MORPH_CLOSE, MORPH_KERNEL, iterations=1)

                # --- 3. 윤곽선 찾기 및 필터링 ---
                contours, _ = cv2.findContours(combined_mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

                self.__signDetectedRegions = []

                for contour in contours:
                    area = cv2.contourArea(contour)
                    if MIN_ROI_AREA < area < MAX_ROI_AREA: # 면적 필터링
                        perimeter = cv2.arcLength(contour, True)
                        approx = cv2.approxPolyDP(contour, 0.04 * perimeter, True) # 다각형 근사화

                        num_vertices = len(approx)
                        x, y, w, h = cv2.boundingRect(contour)

                        # 유효한 너비와 높이인지 확인
                        if w == 0 or h == 0:
                            continue

                        aspect_ratio = float(w) / h

                        # 형태 및 종횡비 필터링
                        # 종횡비는 0.7에서 1.5 정도로 좀 더 유연하게 설정 (다양한 표지판 형태 고려)
                        if (0.7 < aspect_ratio < 1.5): # 일반적인 표지판 종횡비 범위
                            # 꼭짓점 개수로 세부 형태 분류
                            if num_vertices == 3: # 삼각형
                                self.__signDetectedRegions.append((x, y, w, h))
                            elif num_vertices == 4: # 사각형
                                self.__signDetectedRegions.append((x, y, w, h))
                            elif num_vertices == 8: # 팔각형 (정지 표지판)
                                self.__signDetectedRegions.append((x, y, w, h))
                            elif num_vertices > 7 and num_vertices < 15: # 원형에 가까운 다각형
                                # 원형은 HoughCircles로 더 잘 잡히므로, 여기서는 보조적인 역할
                                self.__signDetectedRegions.append((x, y, w, h))

                # --- 4. Hough Circle Transform으로 원형 표지판 추가 감지 ---
                gray = cv2.cvtColor(self.__raw_img, cv2.COLOR_BGR2GRAY)
                blurred_gray = cv2.GaussianBlur(gray, (5, 5), 0)

                circles = cv2.HoughCircles(
                    blurred_gray,
                    cv2.HOUGH_GRADIENT,
                    HOUGH_DP,
                    HOUGH_MIN_DIST,
                    param1=HOUGH_PARAM1,
                    param2=HOUGH_PARAM2,
                    minRadius=HOUGH_MIN_RADIUS,
                    maxRadius=HOUGH_MAX_RADIUS
                )

                if circles is not None:
                    circles = np.uint16(np.around(circles))
                    for i in circles[0, :]:
                        x_center, y_center, r = i[0], i[1], i[2]

                        # 원형 바운딩 박스 계산 (패딩 추가)
                        padding = int(r * 0.2)
                        x_min = max(0, x_center - r - padding)
                        y_min = max(0, y_center - r - padding)
                        x_max = min(self.__raw_img.shape[1], x_center + r + padding)
                        y_max = min(self.__raw_img.shape[0], y_center + r + padding)

                        w = x_max - x_min
                        h = y_max - y_min

                        # 유효한 너비와 높이인지 확인
                        if w > 0 and h > 0:
                            # 중복 방지를 위해 기존 리스트에 이미 비슷한 영역이 있는지 확인하는 로직 추가 가능
                            # 여기서는 간단히 추가
                            self.__signDetectedRegions.append((x_min, y_min, w, h))

                # TODO: 최종적으로 sign_candidate_regions 리스트에 중복된 바운딩 박스가 있다면
                # Non-Maximum Suppression (NMS) 등을 사용하여 제거하는 로직을 추가하면 좋습니다.
                # (이는 더 복잡하므로 필요시 검색하여 구현)



                if self.__signDetectedRegions:
                    print(f"프레임에서 {len(self.__signDetectedRegions)}개의 표지판 후보 영역이 감지되었습니다.")

                    detected_signs = [] # 실제 자른 표지판 이미지들을 저장할 리스트

                    # 찾은 후보 영역 표시 및 처리 (모델 예측 로직을 여기에 넣습니다)
                    for (x, y, w, h) in self.__signDetectedRegions:
                        #cv2.rectangle(self.__raw_img, (x, y), (x + w, y + h), (0, 255, 0), 2) # 초록색 사각형

                        cropped_sign = self.__raw_img[y:y+h, x:x+w]

                        if cropped_sign.shape[0] > 0 and cropped_sign.shape[1] > 0:

#-------------------------------------------------------------------------
                           # predicted_label, confidence = predict_gtsrb_sign(cropped_sign)
#-------------------------------------------------------------------------

                            img_resized = cv2.resize(cropped_sign, (30, 30))

                            # 2. 이미지 픽셀 값 정규화 (0-1)
                            img_normalized = img_resized / 255.0

                            # 3. 배치 차원 추가
                            img_input = np.expand_dims(img_normalized, axis=0) # (1, height, width, channels)

                            # 4. 예측 수행
                            predictions = self.__signModel.predict(img_input)
                            predicted_class_id = np.argmax(predictions[0])
                            confidence = np.max(predictions[0])

                            predicted_label = gtsrb_labels[predicted_class_id]



#-------------------------------------------------------------------------
                            # 결과 표시 (원본 프레임에 오버레이)
                            # 예측 결과와 신뢰도가 특정 임계값 이상일 때만 표시하도록 설정 가능
                            if confidence > 0.7: # 예를 들어, 70% 이상의 신뢰도를 가진 예측만 표시
                                text = f"{predicted_label} ({confidence:.2f})"
                                #cv2.rectangle(self.__raw_img, (x, y), (x + w, y + h), (0, 255, 0), 2) # 초록색 바운딩 박스
                                #cv2.putText(self.__raw_img, text, (x, y - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)
                                print(text)
                            else:
                                # 신뢰도가 낮으면 다른 색으로 표시하거나 무시
                                #cv2.rectangle(self.__raw_img, (x, y), (x + w, y + h), (0, 0, 255), 2) # 빨간색 바운딩 박스
                                #cv2.putText(self.__raw_img, "Unknown/Low Conf", (x, y - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 0, 255), 2)
                                print("Unknown/Low Conf")






            except Exception as e:
                print("Sign detector error : " , e)
                continue

            time.sleep(0.001)
